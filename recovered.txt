444: inline bool split_lognormal_shape_params(double mu, double sigma, double delta,
445:                                          split_lognormal_shape& h) {
446:   if (!(sigma > 0.0) || !emc2_isfinite(mu) || !emc2_isfinite(delta))
447:     return false;
448:   if (delta == 0.0) {
449:     h.c = mu; h.sL = sigma; h.sR = sigma; h.a = 0.5;
450:     return true;
451:   }
452:   h.sL = sigma * std::exp(0.5 * delta);
453:   h.sR = sigma * std::exp(-0.5 * delta);
454:   if (!(h.sL > 0.0) || !(h.sR > 0.0) ||
455:       !emc2_isfinite(h.sL) || !emc2_isfinite(h.sR)) return false;
456:   h.a = 1.0 / (1.0 + std::exp(-delta));
457:   if (delta > 0.0)
458:     h.c = mu - h.sL * R::qnorm(1.0 / (4.0 * h.a), 0.0, 1.0, true, false);
459:   else

461:                                (4.0 * h.sR), 0.0, 1.0, true, false);
